local CopyCorner = Instance.new("UICorner")
CopyCorner.CornerRadius = UDim.new(0, 10)

CopyButton.MouseButton1Click:Connect(function()
    local ok = false
    if setclipboard then
        ok = pcall(setclipboard, GROUP_ID_LINK)
    end

    local originalText = CopyButton.Text
    local originalColor = CopyButton.BackgroundColor3

    if ok then
        CopyButton.Text = "Copied!"
        CopyButton.BackgroundColor3 = Color3.fromRGB(120, 255, 140)
    else
        CopyButton.Text = "Clipboard unsupported"
        CopyButton.BackgroundColor3 = Color3.fromRGB(255, 120, 120)
    end

    task.delay(1.4, function()
        if CopyButton and CopyButton.Parent then
            CopyButton.Text = originalText
            CopyButton.BackgroundColor3 = originalColor
        end
    end)
end)

-- ==== DRAGGING LOGIC ====
local dragging = false
local dragInput
local dragStart
local startPos

local function updateDrag(input)
    local delta = input.Position - dragStart
    Main.Position = UDim2.new(
        startPos.X.Scale, startPos.X.Offset + delta.X,
        startPos.Y.Scale, startPos.Y.Offset + delta.Y
    )
end

TitleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = Main.Position

        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

TitleBar.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        updateDrag(input)
    end
end)
